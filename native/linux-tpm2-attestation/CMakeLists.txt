cmake_minimum_required(VERSION 3.20)
project(cieav_linux_tpm2_attestation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(TSS2_ESYS REQUIRED IMPORTED_TARGET tss2-esys)
pkg_check_modules(TSS2_MU REQUIRED IMPORTED_TARGET tss2-mu)
pkg_check_modules(TSS2_TCTILDR REQUIRED IMPORTED_TARGET tss2-tctildr)
add_executable(cieav-linux-tpm2-attestation main.cpp)
target_compile_options(cieav-linux-tpm2-attestation PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong -D_FORTIFY_SOURCE=2 -fPIE)
target_link_options(cieav-linux-tpm2-attestation PRIVATE -Wl,-z,relro,-z,now -pie)
target_link_libraries(cieav-linux-tpm2-attestation PRIVATE PkgConfig::TSS2_ESYS PkgConfig::TSS2_MU PkgConfig::TSS2_TCTILDR)
