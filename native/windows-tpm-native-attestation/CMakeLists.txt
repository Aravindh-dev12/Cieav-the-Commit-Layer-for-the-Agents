cmake_minimum_required(VERSION 3.20)
project(cieav_windows_tpm_native_attestation LANGUAGES CXX)
add_executable(cieav-windows-tpm-native-attestation main.cpp)
target_compile_features(cieav-windows-tpm-native-attestation PRIVATE cxx_std_20)
target_compile_definitions(cieav-windows-tpm-native-attestation PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(cieav-windows-tpm-native-attestation PRIVATE ncrypt bcrypt crypt32)
if(MSVC)
  target_compile_options(cieav-windows-tpm-native-attestation PRIVATE /W4 /WX /permissive- /guard:cf /sdl)
  target_link_options(cieav-windows-tpm-native-attestation PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
else()
  target_compile_options(cieav-windows-tpm-native-attestation PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-nullability-completeness -fstack-protector-strong -fcf-protection=full)
  target_compile_definitions(cieav-windows-tpm-native-attestation PRIVATE _FORTIFY_SOURCE=2)
  target_link_options(cieav-windows-tpm-native-attestation PRIVATE -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
endif()
install(TARGETS cieav-windows-tpm-native-attestation RUNTIME DESTINATION .)
