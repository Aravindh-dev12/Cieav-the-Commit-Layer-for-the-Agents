cmake_minimum_required(VERSION 3.20)
project(cieav_windows_tpm_measured_boot_probe LANGUAGES CXX)
add_executable(cieav-windows-tpm-measured-boot-probe main.cpp)
target_compile_features(cieav-windows-tpm-measured-boot-probe PRIVATE cxx_std_20)
target_compile_definitions(cieav-windows-tpm-measured-boot-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(cieav-windows-tpm-measured-boot-probe PRIVATE ncrypt bcrypt tbs)
if(MSVC)
  target_compile_options(cieav-windows-tpm-measured-boot-probe PRIVATE /W4 /WX /permissive- /guard:cf /sdl)
  target_link_options(cieav-windows-tpm-measured-boot-probe PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
else()
  target_compile_options(cieav-windows-tpm-measured-boot-probe PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-nullability-completeness -fstack-protector-strong -fcf-protection=full)
  target_compile_definitions(cieav-windows-tpm-measured-boot-probe PRIVATE _FORTIFY_SOURCE=2)
  target_link_options(cieav-windows-tpm-measured-boot-probe PRIVATE -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
endif()
install(TARGETS cieav-windows-tpm-measured-boot-probe RUNTIME DESTINATION .)
