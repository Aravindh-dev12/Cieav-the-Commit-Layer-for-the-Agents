cmake_minimum_required(VERSION 3.20)
project(cieav_windows_tpm_measured_boot LANGUAGES CXX)
add_executable(cieav-windows-tpm-measured-boot main.cpp)
target_compile_features(cieav-windows-tpm-measured-boot PRIVATE cxx_std_20)
target_compile_definitions(cieav-windows-tpm-measured-boot PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(cieav-windows-tpm-measured-boot PRIVATE tbs)
if(MSVC)
  target_compile_options(cieav-windows-tpm-measured-boot PRIVATE /W4 /WX /permissive- /guard:cf /sdl)
  target_link_options(cieav-windows-tpm-measured-boot PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
else()
  target_compile_options(cieav-windows-tpm-measured-boot PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-nullability-completeness -fstack-protector-strong -fcf-protection=full)
  target_compile_definitions(cieav-windows-tpm-measured-boot PRIVATE _FORTIFY_SOURCE=2)
  target_link_options(cieav-windows-tpm-measured-boot PRIVATE -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
endif()
install(TARGETS cieav-windows-tpm-measured-boot RUNTIME DESTINATION .)
