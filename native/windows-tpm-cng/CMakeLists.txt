cmake_minimum_required(VERSION 3.20)
project(cieav_windows_tpm_cng LANGUAGES CXX)
add_executable(cieav-windows-tpm-cng main.cpp)
target_compile_features(cieav-windows-tpm-cng PRIVATE cxx_std_20)
target_compile_definitions(cieav-windows-tpm-cng PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(cieav-windows-tpm-cng PRIVATE ncrypt bcrypt crypt32)
if(MSVC)
  target_compile_options(cieav-windows-tpm-cng PRIVATE /W4 /WX /permissive- /guard:cf /sdl)
  target_link_options(cieav-windows-tpm-cng PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
else()
  target_compile_options(cieav-windows-tpm-cng PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-nullability-completeness -fstack-protector-strong -fcf-protection=full)
  target_compile_definitions(cieav-windows-tpm-cng PRIVATE _FORTIFY_SOURCE=2)
  target_link_options(cieav-windows-tpm-cng PRIVATE -Wl,--dynamicbase -Wl,--nxcompat -Wl,--high-entropy-va)
endif()
